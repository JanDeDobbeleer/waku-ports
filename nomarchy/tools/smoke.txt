# Commands tools\check.ps1 runs against the shell during the runtime step (see check.ps1 for the
# syntax). They switch on the off-by-default plugins that work on Windows and open every menu and
# panel, so the code behind each one runs and logs. Only UI: nothing here changes the machine. The
# plugins the README lists as dropped (they run Linux commands) stay off, as do idle, night light
# and remote session.
call shell setPluginEnabled nomarchy.active-window true => ok
call shell setPluginEnabled nomarchy.media true => ok
call shell setPluginEnabled nomarchy.microphone true => ok
call shell setPluginEnabled nomarchy.spacer true => ok
sleep 3
call shell call nomarchy.menu open {"menu":"root"}
call shell call nomarchy.menu open {"menu":"apps"}
call shell call nomarchy.menu open {"menu":"appearance"}
call shell call nomarchy.menu open {"menu":"appearance.theme"}
sleep 1
call shell call nomarchy.menu open {"menu":"trigger"}
call shell call nomarchy.menu open {"menu":"setup"}
call shell call nomarchy.menu open {"menu":"system"}
call shell hide nomarchy.menu
# The theme picker light-dismisses: summoning the menu over it closes it.
call shell toggle nomarchy.image-picker {"source":"themes"}
sleep 3
call shell call nomarchy.image-picker debugState x => opened.{1,4}true
call shell summon nomarchy.menu {"menu":"root"} => ok
sleep 1
call shell call nomarchy.image-picker debugState x => opened.{1,4}false
call shell hide nomarchy.menu
# Modal overlays are exclusive: summoning one dismisses the others.
call shell summon nomarchy.menu {"menu":"root"} => ok
sleep 2
call shell isOpen nomarchy.menu => open
call shell summon nomarchy.image-picker {"source":"themes"} => ok
sleep 3
call shell isOpen nomarchy.menu => closed
call shell isOpen nomarchy.image-picker => open
call shell hide nomarchy.image-picker
call shell isOpen nomarchy.image-picker => closed
call nomarchy.audio open
call nomarchy.audio close
call nomarchy.bluetooth open
call nomarchy.bluetooth close
call nomarchy.network open
call nomarchy.network close
call nomarchy.power open
call nomarchy.power close
call nomarchy.clock open
call nomarchy.clock close
call nomarchy.weather open
call nomarchy.weather close
call shell toggle nomarchy.dev-gallery {}
call shell toggle nomarchy.dev-gallery {}
call osd show {"kind":"volume","value":0.5}
call osd close
call notifications showHistory
call notifications dismissAll
call shell toggleBarTransparency
call shell toggleBarTransparency
